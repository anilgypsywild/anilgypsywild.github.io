# ---- Canonicalizers / cleaners ----------------------------------------------
# Put these helpers once near the top. They make downstream matching robust to
# curly quotes, NBSP, stray backticks, and spacing issues.
canon_item <- function(x) {
  x <- as.character(x)
  x <- gsub("[\u2018\u2019]", "'", x)      # curly → straight apostrophe
  x <- gsub("^[`′]+|[`′]+$", "", x)       # strip leading/trailing backticks/prime
  x <- gsub("\u00A0", " ", x)             # NBSP → space
  x <- gsub("\\s+", " ", x)               # collapse spaces
  trimws(x)
}

# Header normalizer so grep works despite curly quotes, newlines, odd spaces
normalize_header <- function(x) {
  x <- gsub("[\r\n]+", " ", x)                              # collapse line breaks
  x <- gsub("\u00A0|\u200B|\u200C|\u200D", " ", x)          # NBSP/zero-width
  x <- chartr("\u201C\u201D\u2018\u2019", '""\'\'', x)      # curly -> straight quotes
  x <- gsub("\\s+", " ", x)                                 # collapse spaces
  trimws(x)
}

nms      <- names(raw)
nms_norm <- normalize_header(nms)

# =============================================================================
# 2) Column finders (Listener export style), ignoring "... Duration (ms)" cols
#    Convention in this export:
#      - MOST  = "Question 1: Which of these motivates you the MOST?"
#      - LEAST = "Question 2: Which of these motivates you the LEAST?"
# =============================================================================
keep_mask      <- !grepl("Duration \\(ms\\)$", nms_norm, perl = TRUE)

pat_most  <- '^(\\d+)\\..*\\(Question\\s*1:.*Which of these motivates you the MOST\\??.*\\)\\s*Answer$'
pat_least <- '^(\\d+)\\..*\\(Question\\s*2:.*Which of these motivates you the LEAST\\??.*\\)\\s*Answer$'

worst_i_keep <- grep(pat_least, nms_norm_keep, perl = TRUE)

# Helper: extract leading screen number (e.g., "2." -> 2) for task alignment
lead_num <- function(v) as.integer(sub("^\\s*([0-9]+)\\..*$", "\\1", v))

# Order by screen number and align to intersection of tasks present in BOTH
best_df <- data.frame(
  col      = nms_keep[best_i_keep],
  col_norm = nms_norm_keep[best_i_keep],
  task     = lead_num(nms_norm_keep[best_i_keep]),
  type     = "best",
  stringsAsFactors = FALSE
)

worst_df <- data.frame(
  col      = nms_keep[worst_i_keep],
  col_norm = nms_norm_keep[worst_i_keep],
  task     = lead_num(nms_norm_keep[worst_i_keep]),
  type     = "worst",
  stringsAsFactors = FALSE
)

best_df  <- best_df [order(best_df$task), ]
worst_df <- worst_df[order(worst_df$task), ]

# Keep only tasks that appear in BOTH best and worst

worst_df <- worst_df[worst_df$task %in% common_tasks, ]

# Final column lists (same length)

# 3) Long format: BEST / LEAST per task

  dplyr::mutate(respondent_id = dplyr::row_number()) %>%   # stable row id

    cols = dplyr::all_of(col_key$col),

# Item text cleaner (mild): remove stray backticks and extra spaces
clean_item <- function(x) {
  x <- trimws(x)
  x <- gsub("^`+|`+$", "", x)
  x <- gsub("\\s+", " ", x)
  x
}
long$best  <- clean_item(long$best)
long$worst <- clean_item(long$worst)

# =============================================================================
# 4) Build pairs: each row is a best-over-worst observation
#    - If the export already shipped long format best/worst per task, we use it.
#    - y = 1 encodes "winner beats loser" for binomial logit.
# =============================================================================
build_pairs_row <- function(best, worst, rid, task) {
  data.frame(
    respondent_id = rid,
    task          = task,
    winner        = best,
    loser         = worst,
    y             = 1,                     # best beats worst
    stringsAsFactors = FALSE
  )
}

pairs_all <- do.call(
  rbind,
  lapply(seq_len(nrow(long)), function(i) {
    r <- long[i, ]
    build_pairs_row(r$best, r$worst, r$respondent_id, r$task)
  })
)

message("Pairs built: ", nrow(pairs_all))
utils::head(pairs_all)

# =============================================================================
# 5) Design matrix (difference coding, last item as reference)
#    - Create +1/-1 indicator across items excluding reference.
#    - This matches the theoretical setup for best-worst (difference) logit.
# =============================================================================
items <- sort(unique(c(long$best, long$worst)))
items <- items[!is.na(items) & nzchar(items)]

ref_item  <- tail(items, 1)               # choose last as reference
item_cols <- setdiff(items, ref_item)

# Create one row of X for a (winner, loser) pair
make_X <- function(winner, loser, item_cols) {
  x <- setNames(rep(0, length(item_cols)), item_cols)
  if (!is.na(winner) && winner %in% item_cols) x[winner] <- x[winner] + 1
  if (!is.na(loser)  && loser  %in% item_cols) x[loser ] <- x[loser ] - 1
  x
}

# Build design X and response y
X <- t(apply(pairs_all[, c("winner","loser")], 1, function(r) {
  make_X(r[["winner"]], r[["loser"]], item_cols)
}))
X <- as.data.frame(X, check.names = FALSE)
y <- pairs_all$y

# =============================================================================
# 6) Fit overall model (aggregate binomial logit)
#    - GLM with logit link and no intercept (since reference dropped).
#    - Coefs are utilities for non-reference items; reference is 0 by construction.
# =============================================================================
fit_overall <- glm(y ~ . - 1, family = binomial(), data = cbind(y, X))

co <- coef(fit_overall)
co <- co[!is.na(co)]
names(co) <- clean_item(names(co))

u_hat <- setNames(rep(0, length(items)), items)  # include reference as 0
u_hat[names(co)] <- co
u_hat[ref_item]  <- 0

# Center utilities and scale to 0–100 for interpretability
u_center <- u_hat - mean(u_hat)
rng <- diff(range(u_center))
rel_import <- if (rng > 0) {
  (u_center - min(u_center)) / rng * 100
} else {
  rep(0, length(u_center))
}

# Tidy overall table
overall_table <- data.frame(
  item       = names(u_center),
  utility    = round(as.numeric(u_center), 4),
  importance = round(as.numeric(rel_import), 2),
  stringsAsFactors = FALSE
) %>%
  arrange(desc(importance))

# Vectorized mappers for coarse bins
derive_age_bin_vec <- function(v) {
  chr <- trimws(as.character(v))
  num <- suppressWarnings(as.numeric(chr))
  out <- rep(NA_character_, length(chr))

  # numeric entries (if any)
  out[!is.na(num) & num < 35]  <- "18-34"
  out[!is.na(num) & num >= 35] <- "35+"

  # buckets / ranges in text
  rng_1834 <- grepl("(?i)\\b(18\\s*[–-]\\s*34|25\\s*[–-]\\s*34|18-34|25-34)\\b", chr, perl = TRUE)
  rng_35p  <- grepl("(?i)\\b(35\\s*[–-]\\s*44|45\\s*[–-]\\s*54|55\\s*[–-]\\s*64|65\\+|35-44|45-54|55-64)\\b", chr, perl = TRUE)
  out[is.na(out) & rng_1834] <- "18-34"
  out[is.na(out) & rng_35p]  <- "35+"

  # single ages embedded in free text
  any_1834 <- grepl("\\b(1[89]|2[0-9]|3[0-4])\\b", chr)
  any_35p  <- grepl("\\b(3[5-9]|[4-8][0-9])\\b", chr)
  out[is.na(out) & any_1834] <- "18-34"
  out[is.na(out) & any_35p]  <- "35+"
  out
}

to_donor_type_vec <- function(v) {
  t <- tolower(trimws(as.character(v)))
  out <- rep(NA_character_, length(t))
  hid <- grepl("often|regular|fundrais|active|organis|organize|lead|advocat", t)
  out[hid]                 <- "HID"     # High Involvement Donor (heuristic)
  out[is.na(out) & t != ""] <- "Casual"
  out
}

to_giving_status_vec <- function(v) {
  t <- tolower(trimws(as.character(v)))
  out <- rep(NA_character_, length(t))
  out[grepl("^\\s*(no|not yet|never|first)", t)] <- "FirstTime"
  out[grepl("^\\s*(yes|many|repeat|once)", t)]   <- "Repeat"
  out
}

# =============================================================================
# 8) Segment fitting helper
#    - Fits the same difference-coded logit on a subset of respondents.
#    - Returns a tibble with (item, importance).
# =============================================================================
estimate_segment <- function(idx_named, seg_name, min_rows = 30L) {
  keep_ids <- names(idx_named)[which(!is.na(idx_named) & idx_named)]
  if (length(keep_ids) < min_rows) return(NULL)

  sub <- pairs_all[pairs_all$respondent_id %in% keep_ids, , drop = FALSE]
  if (!nrow(sub)) return(NULL)

  Xs <- t(apply(sub[, c("winner","loser")], 1, function(r) make_X(r[1], r[2], item_cols)))
  Xs <- as.data.frame(Xs, check.names = FALSE)
  ys <- sub$y

  fit <- try(glm(ys ~ . - 1, family = binomial(), data = cbind(ys, Xs),
                 control = list(maxit = 100)), silent = TRUE)
  if (inherits(fit, "try-error")) return(NULL)

  co <- coef(fit); co[is.na(co)] <- 0
  uh <- setNames(rep(0, length(items)), items)
  uh[names(co)] <- co
  uh[ref_item]  <- 0

  ucenter <- uh - mean(uh)
  rng <- max(ucenter) - min(ucenter)
  imp <- if (rng > 0) (ucenter - min(ucenter))/rng*100 else rep(0, length(ucenter))

  tibble::tibble(item = names(imp), importance = as.numeric(imp))
}

# Add small labels on bars
add_value_labels <- function(width = 0.7, size = 3.6, hjust = -0.15, digits = 1) {
  ggplot2::geom_text(
    ggplot2::aes(label = round(after_stat(y), digits)),
    stat = "identity",
    position = ggplot2::position_dodge(width = width),
    hjust = hjust, size = size, color = "#222"
  )
}

# A light, presentation-friendly theme
theme_clean_light <- function(base_size = 13) {
  ggplot2::theme_minimal(base_size = base_size) %+replace%
    ggplot2::theme(
      panel.background = element_rect(fill = "white", colour = NA),
      plot.background  = element_rect(fill = "white", colour = NA),
      panel.grid.major.y = element_blank(),
      panel.grid.minor   = element_blank(),
      axis.title.x = element_text(size = base_size - 1),
      axis.text   = element_text(color = "#222"),
      plot.title  = element_text(face = "bold", hjust = 0.5),
      legend.position = "top",
      legend.title = element_blank(),
      plot.margin = margin(10, 16, 10, 10)
    )
}

# ---- Share of Preference (Overall) -------------------------------------------
plot_share_of_preference <- function(overall_table, outfile, temperature = 1) {
  stopifnot(all(c("item","utility") %in% names(overall_table)))
  if (!dir.exists(dirname(outfile))) dir.create(dirname(outfile), recursive = TRUE)

  # Stable softmax on mean-centered utilities
  u <- as.numeric(overall_table$utility)
  u[is.na(u)] <- 0
  if (!is.finite(temperature) || temperature <= 0) temperature <- 1

  s   <- (u - max(u, na.rm = TRUE)) / temperature
  p   <- exp(s)
  sop <- 100 * p / sum(p)

  df <- overall_table %>%
    dplyr::mutate(share = sop) %>%
    dplyr::arrange(dplyr::desc(share))

  g <- df %>%
    dplyr::mutate(item = forcats::fct_reorder(item, share)) %>%
    ggplot2::ggplot(ggplot2::aes(x = item, y = share)) +
    ggplot2::geom_col(fill = "#5BA3D0", width = 0.65, color = "white") +
    ggplot2::coord_flip(clip = "off") +
    ggplot2::geom_text(ggplot2::aes(label = sprintf("%.1f%%", share)),
                       hjust = -0.15, size = 3.6, color = "#222") +
    ggplot2::scale_y_continuous(expand = ggplot2::expansion(mult = c(0, .08))) +
    ggplot2::labs(title = "Share of Preference (Softmax on Utilities)",
                  x = NULL, y = "Percent") +
    theme_clean_light(14)

  ggplot2::ggsave(outfile, g, width = 9, height = 6.2, dpi = 300)
  invisible(g)
}
plot_share_of_preference(overall_table, "plots/Plot_ShareOfPreference.png")

# ---- Dumbbell delta between two segments -------------------------------------
plot_dumbbell_two_segments <- function(segA, segB, nameA, nameB, outfile,
                                       top_n = 15, label_endpoints = TRUE, wrap = 44) {
  stopifnot(!is.null(segA), !is.null(segB))
  stopifnot(all(c("item","importance") %in% names(segA)),
            all(c("item","importance") %in% names(segB)))

  a <- segA %>% dplyr::select(item, impA = importance) %>%
    dplyr::mutate(item = canon_item(item)) %>%
    dplyr::group_by(item) %>% dplyr::summarise(impA = mean(impA, na.rm = TRUE), .groups = "drop")

  b <- segB %>% dplyr::select(item, impB = importance) %>%
    dplyr::mutate(item = canon_item(item)) %>%
    dplyr::group_by(item) %>% dplyr::summarise(impB = mean(impB, na.rm = TRUE), .groups = "drop")

  df <- dplyr::inner_join(a, b, by = "item") %>%
    dplyr::mutate(delta = impA - impB) %>%
    dplyr::arrange(dplyr::desc(abs(delta))) %>%
    dplyr::slice_head(n = top_n)

  if (!nrow(df)) {
    warning("No overlapping items between segments; nothing to plot.")
    return(invisible(NULL))
  }

  df <- df %>% dplyr::mutate(order_key = pmax(impA, impB))
  df$item <- forcats::fct_reorder(df$item, df$order_key, .desc = TRUE)

  # palette fallback
  if (!exists("pal_light")) {
    pal_light <<- c(
      "NextGen"            = "#5BA3D0",
      "CurrentGen"         = "#9AC27E",
      "High impact donors" = "#7F8AC6",
      "Casual supporters"  = "#F2A640",
      "First time"         = "#71C6B1",
      "Repeat"             = "#D77C8C"
    )
  }

  g <- ggplot2::ggplot(df, ggplot2::aes(y = stringr::str_wrap(item, width = wrap))) +
    ggplot2::geom_segment(
      ggplot2::aes(yend = stringr::str_wrap(item, width = wrap), x = impB, xend = impA),
      linewidth = 1.2, color = "#e0e0e0"
    ) +
    ggplot2::geom_point(ggplot2::aes(x = impB, color = nameB), size = 3) +
    ggplot2::geom_point(ggplot2::aes(x = impA, color = nameA), size = 3) +
    ggplot2::scale_color_manual(
      values = pal_light[c(nameA, nameB)],
      breaks = c(nameA, nameB), limits = c(nameA, nameB), name = NULL
    ) +
    ggplot2::labs(
      title = paste0(nameA, " vs ", nameB, " — \u0394 Importance"),
      x = "Importance (0–100)", y = NULL
    ) +
    ggplot2::scale_x_continuous(expand = ggplot2::expansion(mult = c(0.02, 0.06))) +
    theme_clean_light(13)

  if (isTRUE(label_endpoints)) {
    g <- g +
      ggplot2::geom_text(ggplot2::aes(x = impB, label = round(impB, 1)),
                         nudge_y = 0.15, size = 3, color = "#333") +
      ggplot2::geom_text(ggplot2::aes(x = impA, label = round(impA, 1)),
                         nudge_y = -0.15, size = 3, color = "#333")
  }

  ggplot2::ggsave(outfile, g, width = 9.5, height = 6.8, dpi = 300)
  invisible(g)
}

# =============================================================================
# 10) Bootstrap CIs for Overall Importance
#     - Nonparametric respondent-level bootstrap -> 95% percentile CI.
#     - Defensive coding: handle singular fits by substituting zeros on failure.
# =============================================================================
bootstrap_overall <- function(pairs_all, items, ref_item, B = 200, seed = 123) {
  stopifnot(all(c("respondent_id","winner","loser","y") %in% names(pairs_all)))
  set.seed(seed)

  items     <- clean_item(items)
  ref_item  <- clean_item(ref_item)
  item_cols <- setdiff(items, ref_item)

  make_X <- function(winner, loser, item_cols) {
    x <- rep(0, length(item_cols)); names(x) <- item_cols
    if (!is.na(winner) && winner %in% item_cols) x[winner] <- x[winner] + 1
    if (!is.na(loser)  && loser  %in% item_cols) x[loser ] <- x[loser ] - 1
    x
  }

  fit_one <- function(df) {
    Xs <- t(apply(df[, c("winner","loser")], 1, function(r) make_X(r[1], r[2], item_cols)))
    Xs <- as.data.frame(Xs, check.names = FALSE)
    ys <- df$y
    co <- tryCatch(
      coef(glm(ys ~ . - 1, family = binomial(), data = cbind(ys, Xs),
               control = list(maxit = 100))),
      error = function(e) rep(NA_real_, length(item_cols))
    )
    co[is.na(co)] <- 0
    names(co) <- clean_item(names(co))

    uh <- setNames(rep(0, length(items)), items)
    uh[names(co)] <- co
    uh[ref_item]  <- 0

    ucenter <- uh - mean(uh)
    rng <- max(ucenter) - min(ucenter)
    if (rng > 0) (ucenter - min(ucenter))/rng*100 else rep(0, length(ucenter))
  }

  ids <- unique(pairs_all$respondent_id)
  M   <- length(ids)
  mat <- matrix(NA_real_, nrow = B, ncol = length(items))
  colnames(mat) <- items

  for (b in seq_len(B)) {
    samp_ids <- sample(ids, size = M, replace = TRUE)
    sub <- pairs_all[pairs_all$respondent_id %in% samp_ids, , drop = FALSE]
    mat[b, ] <- fit_one(sub)
  }

  tibble::tibble(
    item = items,
    lo   = apply(mat, 2, function(x) stats::quantile(x, 0.025, na.rm = TRUE)),
    hi   = apply(mat, 2, function(x) stats::quantile(x, 0.975, na.rm = TRUE))
  )
}

plot_overall_ci <- function(overall_table, ci_df, outfile) {
  df <- dplyr::left_join(
          overall_table %>% dplyr::mutate(item = clean_item(item)),
          ci_df %>% dplyr::mutate(item = clean_item(item)),
          by = "item"
        ) %>%
        dplyr::mutate(item = forcats::fct_reorder(item, importance))

  g <- ggplot2::ggplot(df, ggplot2::aes(x = item, y = importance)) +
    ggplot2::geom_col(fill = "#5BA3D0", width = 0.65, color = "white") +
    ggplot2::geom_errorbar(ggplot2::aes(ymin = lo, ymax = hi), linewidth = 0.6, width = 0.15) +
    ggplot2::coord_flip(clip = "off") +
    ggplot2::scale_y_continuous(expand = ggplot2::expansion(mult = c(0, .08))) +
    ggplot2::labs(title = "Overall Relative Importance (Bootstrap 95% CI)",
                  x = NULL, y = "Importance (0–100)") +
    theme_clean_light(14)

  ggplot2::ggsave(outfile, g, width = 9.5, height = 6.5, dpi = 300)
  invisible(g)
}

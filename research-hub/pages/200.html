#!/usr/bin/env python
# -*- coding: utf-8 -*-

# Thailand FCP UX analysis – TLX, SUS, CSAT, NPS

import pandas as pd
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf

# -------------------------------------------------------------------
# 0. Global plotting style
# -------------------------------------------------------------------
sns.set_theme(style="whitegrid", context="talk", palette="colorblind")

# -------------------------------------------------------------------
# 1. Load and clean data
# -------------------------------------------------------------------
df = pd.read_csv("data/AI_assistant_FCP_THVisit.csv")

# keep only rows where Day starts with "Day"
df_clean = df[df["Day"].astype(str).str.startswith("Day")].copy()

# -------------------------------------------------------------------
# 2. Core metrics: TLX, CSAT, NPS, UMUX, SUS
# -------------------------------------------------------------------

# TLX mental demand (Q1) scaled 0–100
df_clean["TLX"] = (df_clean["Q1"].astype(float) - 1) / 6 * 100

# CSAT from Q5 (1–7)
df_clean["CSAT"] = df_clean["Q5"].astype(float)

# NPS item from Q6 (1–7)
df_clean["NPS_raw"] = df_clean["Q6"].astype(float)

def nps_group(x):
    if x <= 4:
        return "Detractor"
    elif x == 5:
        return "Passive"
    else:
        return "Promoter"

df_clean["NPS_group"] = df_clean["NPS_raw"].apply(nps_group)

# UMUX and SUS from Q2, Q3
df_clean["UMUX_raw"] = df_clean[["Q2", "Q3"]].astype(float).mean(axis=1)
df_clean["UMUX_100"] = (df_clean["UMUX_raw"] - 1) / 6 * 100
df_clean["SUS"] = 0.65 * df_clean["UMUX_100"] + 22.9

# day order
day_order = sorted(df_clean["Day"].unique())

# -------------------------------------------------------------------
# 3. Helper: save-figure wrapper
# -------------------------------------------------------------------
def savefig(name):
    plt.savefig(name, dpi=300, bbox_inches="tight")

# -------------------------------------------------------------------
# 4. NPS distribution
# -------------------------------------------------------------------
plt.figure(figsize=(8, 5))
ax = sns.histplot(
    data=df_clean,
    x="NPS_raw",
    bins=[0.5, 1.5, 2.5, 3.5, 4.5, 5.5, 6.5, 7.5],
    edgecolor="black",
)
mean_nps = df_clean["NPS_raw"].mean()
ax.axvline(mean_nps, linestyle="--", linewidth=2, label=f"Mean = {mean_nps:.2f}")
ax.set_xticks(range(1, 8))
ax.set_xlabel("NPS item rating (1 = very unlikely, 7 = very likely)")
ax.set_ylabel("Number of participants")
ax.set_title("Likelihood to recommend workflow (NPS item)")
ax.legend(loc="upper left")
plt.tight_layout()
savefig("01_nps_distribution.png")
plt.show()

# -------------------------------------------------------------------
# 5. SUS by day (box + jitter)
# -------------------------------------------------------------------
plt.figure(figsize=(10, 5))
ax = sns.boxplot(
    data=df_clean,
    x="Day",
    y="SUS",
    order=day_order,
    showfliers=False,
)
sns.stripplot(
    data=df_clean,
    x="Day",
    y="SUS",
    order=day_order,
    color="black",
    alpha=0.7,
    ax=ax,
)
ax.axhline(68, color="grey", linestyle="--", linewidth=1, label="OK threshold (68)")
ax.axhline(80, color="grey", linestyle=":", linewidth=1, label="Excellent (~80)")
ax.set_title("System Usability Scale (SUS) by day")
ax.set_xlabel("Study day")
ax.set_ylabel("SUS score (0–100)")
ax.legend(loc="lower right")
plt.tight_layout()
savefig("02_sus_by_day.png")
plt.show()

# -------------------------------------------------------------------
# 6. TLX mental demand by day (box + jitter)
# -------------------------------------------------------------------
plt.figure(figsize=(10, 5))
ax = sns.boxplot(
    data=df_clean,
    x="Day",
    y="TLX",
    order=day_order,
    showfliers=False,
)
sns.stripplot(
    data=df_clean,
    x="Day",
    y="TLX",
    order=day_order,
    color="black",
    alpha=0.7,
    ax=ax,
)
ax.axhline(33, color="grey", linestyle="--", linewidth=1, label="Low / Medium")
ax.axhline(66, color="grey", linestyle=":", linewidth=1, label="Medium / High")
ax.set_title("NASA TLX mental demand by day")
ax.set_xlabel("Study day")
ax.set_ylabel("Mental workload (0–100)")
ax.legend(loc="upper right")
plt.tight_layout()
savefig("03_tlx_by_day.png")
plt.show()

# -------------------------------------------------------------------
# 7. CSAT by day (box + jitter)
# -------------------------------------------------------------------
plt.figure(figsize=(10, 5))
ax = sns.boxplot(
    data=df_clean,
    x="Day",
    y="CSAT",
    order=day_order,
    showfliers=False,
)
sns.stripplot(
    data=df_clean,
    x="Day",
    y="CSAT",
    order=day_order,
    color="black",
    alpha=0.7,
    ax=ax,
)
ax.set_title("CSAT (likelihood of using workflow again) by day")
ax.set_xlabel("Study day")
ax.set_ylabel("CSAT rating (1–7)")
plt.tight_layout()
savefig("04_csat_by_day.png")
plt.show()

# -------------------------------------------------------------------
# 8. Heatmap: TLX + SUS by day
# -------------------------------------------------------------------
day_summary = df_clean.groupby("Day").agg(
    {"TLX": "mean", "SUS": "mean"}
).T
day_summary.index = ["Mental Workload (TLX)", "Usability (SUS)"]

# invert TLX for color mapping
plot_summary = day_summary.copy()
plot_summary.loc["Mental Workload (TLX)"] = 100 - plot_summary.loc["Mental Workload (TLX)"]

plt.figure(figsize=(12, 4))
ax = sns.heatmap(
    plot_summary,
    annot=day_summary,
    fmt=".1f",
    cmap="RdYlGn",
    linewidths=0.5,
    linecolor="white",
    cbar_kws={"label": "Better  \u2190  Score  \u2192  Worse"},
)
ax.set_title("Mental Workload and Usability by Day")
ax.set_xlabel("Study Day")
ax.set_ylabel("Metric")
plt.xticks(rotation=0)
plt.yticks(rotation=0)
plt.tight_layout()
savefig("05_tlx_vs_sus_heatmap.png")
plt.show()

# -------------------------------------------------------------------
# 9. Heatmap: TLX + NPS by day
# -------------------------------------------------------------------
def compute_nps_score(sub):
    if len(sub) == 0:
        return np.nan
    promoters = (sub["NPS_group"] == "Promoter").mean()
    detractors = (sub["NPS_group"] == "Detractor").mean()
    return (promoters - detractors) * 100

nps_by_day = df_clean.groupby("Day", group_keys=False).apply(compute_nps_score)

matrix = pd.DataFrame(
    {
        "TLX": df_clean.groupby("Day")["TLX"].mean(),
        "NPS": nps_by_day,
    }
).T
matrix.index = ["Mental Workload (TLX)", "NPS Score"]

plot_matrix = matrix.copy()
plot_matrix.loc["Mental Workload (TLX)"] = 100 - plot_matrix.loc["Mental Workload (TLX)"]

plt.figure(figsize=(12, 4))
ax = sns.heatmap(
    plot_matrix,
    annot=matrix,
    fmt=".1f",
    cmap="RdYlGn",
    linewidths=0.5,
    linecolor="white",
    cbar_kws={"label": "Better  \u2190  Score  \u2192  Worse"},
)
ax.set_title("TLX Mental Workload and NPS by Day")
ax.set_xlabel("Study Day")
ax.set_ylabel("Metric")
plt.xticks(rotation=0)
plt.yticks(rotation=0)
plt.tight_layout()
savefig("06_tlx_vs_nps_heatmap.png")
plt.show()

# -------------------------------------------------------------------
# 10. 4-panel histograms: NPS, CSAT, SUS, TLX
# -------------------------------------------------------------------
fig, axes = plt.subplots(2, 2, figsize=(14, 10))

# NPS
sns.histplot(
    df_clean["NPS_raw"],
    bins=np.arange(0.5, 8.5, 1),
    edgecolor="black",
    ax=axes[0, 0],
)
axes[0, 0].set_title("Likelihood to recommend the workflow (NPS item)")
axes[0, 0].set_xlabel("NPS item rating (1–7)")
axes[0, 0].set_ylabel("Count")

# CSAT
sns.histplot(
    df_clean["CSAT"],
    bins=np.arange(0.5, 8.5, 1),
    edgecolor="black",
    color="teal",
    ax=axes[0, 1],
)
axes[0, 1].set_title("Likelihood to use the workflow again (CSAT)")
axes[0, 1].set_xlabel("CSAT rating (1–7)")
axes[0, 1].set_ylabel("Count")

# SUS
sns.histplot(
    df_clean["SUS"],
    bins=10,
    edgecolor="black",
    color="purple",
    ax=axes[1, 0],
)
axes[1, 0].set_title("Perceived usability (SUS)")
axes[1, 0].set_xlabel("SUS score (0–100)")
axes[1, 0].set_ylabel("Count")

# TLX
sns.histplot(
    df_clean["TLX"],
    bins=10,
    edgecolor="black",
    color="darkred",
    ax=axes[1, 1],
)
axes[1, 1].set_title("Cognitive workload (NASA TLX)")
axes[1, 1].set_xlabel("Mental demand (0–100)")
axes[1, 1].set_ylabel("Count")

plt.tight_layout()
savefig("07_ux_hist_4panel.png")
plt.show()

# -------------------------------------------------------------------
# 11. TLX vs NPS / CSAT / SUS regressions (scatter + line)
# -------------------------------------------------------------------
fig, axes = plt.subplots(2, 2, figsize=(14, 12))

sns.regplot(
    data=df_clean,
    x="TLX",
    y="NPS_raw",
    ax=axes[0, 0],
    scatter_kws={"alpha": 0.7},
)
axes[0, 0].set_title("Workload vs recommendation (TLX → NPS)")
axes[0, 0].set_xlabel("Mental workload (TLX 0–100)")
axes[0, 0].set_ylabel("NPS item (1–7)")

sns.regplot(
    data=df_clean,
    x="TLX",
    y="CSAT",
    ax=axes[0, 1],
    scatter_kws={"alpha": 0.7},
    color="teal",
)
axes[0, 1].set_title("Workload vs likelihood of reuse (TLX → CSAT)")
axes[0, 1].set_xlabel("Mental workload (TLX 0–100)")
axes[0, 1].set_ylabel("CSAT (1–7)")

sns.regplot(
    data=df_clean,
    x="TLX",
    y="SUS",
    ax=axes[1, 0],
    scatter_kws={"alpha": 0.7},
    color="purple",
)
axes[1, 0].set_title("Workload vs perceived usability (TLX → SUS)")
axes[1, 0].set_xlabel("Mental workload (TLX 0–100)")
axes[1, 0].set_ylabel("SUS (0–100)")

axes[1, 1].axis("off")

plt.tight_layout()
savefig("08_tlx_vs_nps_csat_sus_4panel.png")
plt.show()

# -------------------------------------------------------------------
# 12. UX 4x4 scorecard + correlation heatmap
# -------------------------------------------------------------------
metrics = ["TLX", "NPS_raw", "CSAT", "SUS"]

scorecard = df_clean[metrics].agg(["mean", "std", "min", "max"]).T
scorecard.index = [
    "Mental workload (TLX 0–100)",
    "Likelihood to recommend (NPS item 1–7)",
    "Likelihood to reuse (CSAT 1–7)",
    "Usability (SUS 0–100)",
]
print("\nUX 4x4 Scorecard:")
print(scorecard.round(2))

corr_mat = df_clean[metrics].corr()

plt.figure(figsize=(8, 6))
ax = sns.heatmap(
    corr_mat,
    annot=True,
    fmt=".2f",
    cmap="RdYlGn",
    linewidths=0.5,
    linecolor="white",
    vmin=-1,
    vmax=1,
    cbar_kws={"label": "Pearson correlation (r)"},
)
ax.set_title("Workload, usability and satisfaction\n(TLX, NPS, CSAT, SUS)")
plt.tight_layout()
savefig("09_ux_corr_heatmap.png")
plt.show()

# -------------------------------------------------------------------
# 13. Simple regression: NPS ~ CSAT (best model from earlier)
# -------------------------------------------------------------------
model_nps = smf.ols("NPS_raw ~ CSAT", data=df_clean).fit()
print("\nNPS ~ CSAT regression summary:")
print(model_nps.summary())
